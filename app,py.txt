from flask import Flask,render_template,request,redirect,url_for,Response
import cv2
import imutils

from tensorflow.keras.models import model_from_json
import numpy as np
import time
from rembg import remove
from PIL import Image
from cvzone.HandTrackingModule import HandDetector
import math
import os
import time

def predict_img(model,frame):
    return model.predict(frame)

json_file = open('new/model.json', 'r')
loaded_model_json = json_file.read()
json_file.close()
model = model_from_json(loaded_model_json)
model.load_weights("new/model.h5")
print("Loaded model from disk")
print("[Info] starting video")
pred_list = ['a','b','c','d','e','f','g','h','i','j','k','l','m','n','o','p','q','r','s','t','u','v','w','x','y','z']
word = ""
detector = HandDetector(maxHands=1)
app = Flask(__name__)
camera = cv2.VideoCapture(0)
imgHand = []
sentence = ""
def getFrames():
    global imgHand
    while True:
        s,frame = camera.read()
        if not s:
            break
        else:
            
            hands,frame  = detector.findHands(frame)
            imgHand = []
            
            if hands:
                hand = hands[0]
                x,y,w,h = hand["bbox"]
                imgHand = frame[y-20:y+h+20,x-20:x+w+20]
            r,buffer = cv2.imencode('.jpg',frame)
            frame = buffer.tobytes()
        yield(b'--frame\r\n'b'Content-Type: image/jpeg\r\n\r\n' + frame + b'\r\n')
            
@app.route("/video_stream",methods=['GET','POST'])
def video_stream():
    return Response(getFrames(),mimetype='multipart/x-mixed-replace; boundary=frame')



@app.route("/index",methods=['GET','POST'])
def index():
    global sentence
    sentence = ""
    return render_template("index.html",message="")
     
@app.route("/predict_char",methods=['GET','POST'])
def predict_char():
        global imgHand
        global sentence
        imgHand=cv2.resize(imgHand,(300,300))
        imgHand=img_to_array(imgHand)
        imgHand = imgHand.reshape(1,300,300,3)
        pred = predict_img(model,imgHand)
        ch = str(pred_list[np.argmax(pred)])
        sentence = sentence + ch
        return sentence
@app.route("/addDis",methods=['GET','POST'])
def addDis():
        global sentence
        sentence = sentence + " "
        return sentence
@app.route("/delete_char",methods=['GET','POST'])
def delete_char():
        global sentence
        sentence = sentence[:-1]
        return sentence
@app.route("/close_cam",methods=['GET','POST'])
def close_cam():
        global sentence
        sentence = ""
        return sentence

    
if __name__ == "__main__":
    app.run()
